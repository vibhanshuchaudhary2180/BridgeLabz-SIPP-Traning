INTERFACE Searchable
    METHOD search(keyword)
END INTERFACE

INTERFACE Printable
    METHOD print()
END INTERFACE


CLASS PatientReport IMPLEMENTS Searchable, Printable
    VARIABLE patientName
    VARIABLE diagnosis

    METHOD search(keyword):
        IF keyword matches patientName OR diagnosis:
            PRINT "Match found in PatientReport"
        ELSE:
            PRINT "No match"
    
    METHOD print():
        PRINT "Patient Report: " + patientName + ", Diagnosis: " + diagnosis
END CLASS


CLASS LabReport IMPLEMENTS Searchable, Printable
    VARIABLE testName
    VARIABLE result

    METHOD search(keyword):
        IF keyword matches testName OR result:
            PRINT "Match found in LabReport"
        ELSE:
            PRINT "No match"
    
    METHOD print():
        PRINT "Lab Report: " + testName + " = " + result
END CLASS


MAIN:
    CREATE report1 = new PatientReport("Amit", "Fever")
    CREATE report2 = new LabReport("Blood Test", "Normal")

    // Polymorphism
    DECLARE reports[ ] = { report1, report2 }

    FOR EACH report IN reports:
        report.print()
        report.search("Fever")
    END FOR
END MAIN
